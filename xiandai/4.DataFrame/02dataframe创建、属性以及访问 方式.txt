2.loc是通过位置索引,iloc是通过下角标索引，
import pandas as pd
data = {
    '姓名': ['小明', '小红', '小刚'],
    '年龄': [20, 18, 22],
    '成绩': [85, 90, 88]
}
df = pd.DataFrame(data)
print(df)
print(df.loc[0, '姓名'])
print(df.loc[0:2, '姓名':'成绩'])
print(df.iloc[0, 0])
print(df.iloc[0:1, 0:1])
